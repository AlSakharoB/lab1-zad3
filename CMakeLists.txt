cmake_minimum_required(VERSION 3.20)
project(matrix_arithmetic_benchmark VERSION 2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "This benchmark requires GCC and an x86-64 CPU with AVX2")
endif()
set(default_optimization_flags "-O3;-march=native")
set(MATRIX_OPTIMIZATION_FLAGS "${default_optimization_flags}" CACHE STRING "Optimized kernel flags")
option(BUILD_STREAMING "Build the additional AVX streaming-store variant" OFF)
set(PROJECT_WARNING_OPTIONS -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
# Only the arithmetic kernel changes optimization level between variants.
add_library(matrix_support STATIC src/matrix.cpp src/validation.cpp)
target_include_directories(matrix_support PUBLIC include)
target_compile_options(matrix_support PRIVATE ${PROJECT_WARNING_OPTIONS} -O3)

function(add_benchmark_variant target_name variant_name implementation)
    add_library(${target_name}_driver OBJECT src/main.cpp)
    target_include_directories(${target_name}_driver PRIVATE include)
    target_compile_options(${target_name}_driver PRIVATE ${PROJECT_WARNING_OPTIONS} -O3)
    target_compile_definitions(${target_name}_driver PRIVATE
        MATRIX_ARITHMETIC_VARIANT_NAME="${variant_name}"
        MATRIX_ARITHMETIC_DEFAULT_IMPLEMENTATION=${implementation})
    add_executable(${target_name} $<TARGET_OBJECTS:${target_name}_driver> src/arithmetic.cpp)
    target_link_libraries(${target_name} PRIVATE matrix_support)
    target_compile_options(${target_name} PRIVATE ${PROJECT_WARNING_OPTIONS})
endfunction()
add_benchmark_variant(benchmark_baseline baseline 0)
target_compile_options(benchmark_baseline PRIVATE -O0 -fno-tree-vectorize -fno-tree-slp-vectorize)
add_benchmark_variant(benchmark_compiler compiler_optimized 0)
target_compile_options(benchmark_compiler PRIVATE ${MATRIX_OPTIMIZATION_FLAGS})
add_benchmark_variant(benchmark_simd simd 1)
target_compile_options(benchmark_simd PRIVATE ${MATRIX_OPTIMIZATION_FLAGS} -fopenmp-simd)
target_compile_definitions(benchmark_simd PRIVATE MATRIX_ARITHMETIC_ENABLE_SIMD=1)
set(BENCHMARK_VARIANTS baseline compiler simd)
if(BUILD_STREAMING)
    add_benchmark_variant(benchmark_streaming streaming 2)
    target_compile_options(benchmark_streaming PRIVATE ${MATRIX_OPTIMIZATION_FLAGS} -fopenmp-simd)
    target_compile_definitions(benchmark_streaming PRIVATE MATRIX_ARITHMETIC_ENABLE_SIMD=1)
    list(APPEND BENCHMARK_VARIANTS streaming)
endif()

include(CTest)
if(BUILD_TESTING)
    foreach(variant IN LISTS BENCHMARK_VARIANTS)
        set(test_target tests_${variant})
        add_executable(${test_target} tests/test_main.cpp src/arithmetic.cpp)
        target_link_libraries(${test_target} PRIVATE matrix_support)
        get_target_property(variant_options benchmark_${variant} COMPILE_OPTIONS)
        get_target_property(variant_definitions benchmark_${variant} COMPILE_DEFINITIONS)
        target_compile_options(${test_target} PRIVATE ${variant_options})
        if(variant_definitions)
            target_compile_definitions(${test_target} PRIVATE ${variant_definitions})
        endif()
        add_test(NAME ${test_target} COMMAND ${test_target})
        add_test(NAME cli_${variant}_both COMMAND benchmark_${variant}
            --rows 3 --cols 13 --repeats 2 --warmup 0 --iterations 2 --min-time-ms 0 --csv-header)
        foreach(invalid IN ITEMS size repeats iterations operation)
            add_test(NAME cli_${variant}_invalid_${invalid} COMMAND benchmark_${variant} --${invalid} 0)
            set_tests_properties(cli_${variant}_invalid_${invalid} PROPERTIES WILL_FAIL TRUE)
        endforeach()
        foreach(invalid_time IN ITEMS nan inf -1)
            add_test(NAME cli_${variant}_invalid_time_${invalid_time}
                COMMAND benchmark_${variant} --min-time-ms ${invalid_time})
            set_tests_properties(cli_${variant}_invalid_time_${invalid_time} PROPERTIES WILL_FAIL TRUE)
        endforeach()
    endforeach()
endif()
